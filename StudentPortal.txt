// Student data structure
// used for storing basic student info for now, will be updated later

typedef struct {
    int id;
    char name[100];       // increased size to fit longer names
    int age;              // student's age
    float cgpa;           // cumulative GPA
    char department[50];  // e.g CSE, EEE, BBA
    char email[100];      // contact email
    char phone[20];       // contact number
    char address[100];    // home/mailing address
    char gender[10];      // e.g Male, Female
    int semester;         // current semester number
    int creditHours;      // total credit hours completed so far
    char status[20];      // Active / Inactive / Graduated
} Student;
